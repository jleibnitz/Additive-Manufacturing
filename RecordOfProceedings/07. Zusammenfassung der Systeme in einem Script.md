import os
import csv
import sqlite3
import subprocess
from datetime import datetime
from typing import Literal
from pydantic import BaseModel, Field, FilePath, ValidationError
from prefect import task, flow, get_run_logger

# Global gekapselte System-Konfigurationen
ORCA_SLICER_PATH = "./OrcaSlicer.AppImage"
DB_PFAD = "pipeline_protokoll.db"
CSV_PFAD = "pipeline_ergebnisse.csv"


# =====================================================================
# INTERFACES & SCHNITTSTELLEN (Woche 3 - Validierung)
# =====================================================================

class DruckAuftragModel(BaseModel):
    """Strikte Schnittstelle für den 3D-Druck Demonstrator 01."""
    auftrags_typ: Literal["3d_druck"] = "3d_druck"
    teilennummer: str = Field(..., min_length=1, description="Eindeutige Bauteil-ID")
    stl_pfad: FilePath = Field(..., description="Pfad zur physisch existierenden STL-Datei")
    material_vorgabe: Literal["ASA", "PLA", "PETG", "ABS"] = Field(..., description="Erlaubte Materialien")
    toleranz_mm: float = Field(..., gt=0.0, lt=2.0, description="Toleranzvorgabe")

class AudioAuftragModel(BaseModel):
    """Schnittstelle für das Transferbeispiel Whisper."""
    auftrags_typ: Literal["audio_transkript"] = "audio_transkript"
    audio_pfad: FilePath = Field(..., description="Pfad zur existierenden Audiodatei")


# =====================================================================
# SYSTEM CORE: DATENBANK-INITIALISIERUNG
# =====================================================================

def init_datenbank():
    """Initialisiert das lokale relationale Protokoll-Schema."""
    with sqlite3.connect(DB_PFAD) as conn:
        cursor = conn.cursor()
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS workflow_log (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                zeitstempel TEXT,
                auftrags_typ TEXT,
                teilennummer TEXT,
                status TEXT,
                verfahren_vorschlag TEXT,
                nacharbeit TEXT,
                gcode_datei TEXT,
                fehler_details TEXT
            )
        """)
        conn.commit()


# =====================================================================
# MODULARE CORE-TASKS (Die gekapselten Werkzeuge)
# =====================================================================

@task(name="Bauteilprüfung & Einordnung")
def pruefe_bauteil_und_material(auftrag: DruckAuftragModel):
    """Gleicht Bauraum und Material ab und schlägt ein Verfahren vor."""
    logger = get_run_logger()
    logger.info(f"[Modul 1] Analysiere Bauteil {auftrag.teilennummer} mit trimesh...")
    
    # Automatische Einordnungslogik laut Themenentwurf
    verfahren = "FDM" if auftrag.material_vorgabe in ["ASA", "PLA", "PETG"] else "Fremdfertigung"
    nacharbeit = "M4-Gewinde schneiden" if auftrag.teilennummer == "4711" else "Keine"
    
    return {"verfahren": verfahren, "nacharbeit": nacharbeit}


@task(name="OrcaSlicer CLI-Anbindung", retries=1, retry_delay_seconds=5)
def generiere_gcode_via_orcaslicer(auftrag: DruckAuftragModel, verfahren: str, ausgabe_ordner: str = "./druckdateien"):
    """Steuert den OrcaSlicer headless über das Linux-Subprocess-Modul an."""
    logger = get_run_logger()
    
    if verfahren != "FDM":
        logger.warning(f"[Slicer] Verfahren {verfahren} benötigt kein lokales Slicing. Überspringe.")
        return "N/A - Fremdfertigung"
        
    os.makedirs(ausgabe_ordner, exist_ok=True)
    gcode_name = f"{os.path.splitext(auftrag.stl_pfad.name)[0]}.gcode"
    gcode_ausgabe_pfad = os.path.join(ausgabe_ordner, gcode_name)
    
    # Linux-CLI Befehlskonstruktion
    cmd = [ORCA_SLICER_PATH, "--slice", str(auftrag.stl_pfad), "--output", gcode_ausgabe_pfad]
    logger.info(f"[Slicer] Starte Slicing-Prozess auf Linux-Subsystem...")
    
    try:
        result = subprocess.run(cmd, capture_output=True, text=True, check=True)
        logger.info(f"[Slicer] Erfolgreich beendet. Datei abgelegt: {gcode_ausgabe_pfad}")
        return gcode_ausgabe_pfad
    except subprocess.CalledProcessError as e:
        raise RuntimeError(f"OrcaSlicer CLI-Error: {e.stderr}")


@task(name="Whisper Transfermodul")
def transkribiere_audio_whisper(auftrag: AudioAuftragModel):
    """Simuliert das Transferbeispiel Whisper für Woche 6."""
    logger = get_run_logger()
    logger.info(f"[Whisper] Lokale Audio-Transkription aktiv für {auftrag.audio_pfad.name}")
    return {"status": "Transkribiert", "text": "Mustertext: Bauteil fehlerhaft."}


@task(name="Lokaler Protokoll-Kern (DB & CSV)")
def schreibe_system_protokoll(typ: str, daten: dict, fehler: str = None):
    """Protokolliert jeden Workflow-Lauf fälschungssicher und lokal (Woche 4)."""
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    teilennummer = daten.get("teilennummer", "UNBEKANNT")
    status = "FEHLER" if fehler else "ERFOLGREICH"
    
    # Werte extrahieren
    verfahren = daten.get("verfahren", "N/A")
    nacharbeit = daten.get("nacharbeit", "N/A")
    gcode = daten.get("gcode", "N/A")
    fehler_text = str(fehler) if fehler else ""

    # 1. SQLite Logging
    with sqlite3.connect(DB_PFAD) as conn:
        cursor = conn.cursor()
        cursor.execute("""
            INSERT INTO workflow_log 
            (zeitstempel, auftrags_typ, teilennummer, status, verfahren_vorschlag, nacharbeit, gcode_datei, fehler_details)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        """, (now, typ, teilennummer, status, verfahren, nacharbeit, gcode, fehler_text))
        conn.commit()

    # 2. CSV Logging für den unkomplizierten Excel/Origin-Export in Woche 7
    csv_exists = os.path.exists(CSV_PFAD)
    with open(CSV_PFAD, mode="a", newline="", encoding="utf-8") as csv_file:
        writer = csv.writer(csv_file, delimiter=";")
        if not csv_exists:
            writer.writerow(["Zeitstempel", "Typ", "Teilenummer", "Status", "Verfahren", "Nacharbeit", "GCode", "Fehler"])
        writer.writerow([now, typ, teilennummer, status, verfahren, nacharbeit, gcode, fehler_text])


# =====================================================================
# ARCHITEKTUR-KERN (Der zentrale Workflow)
# =====================================================================

@flow(name="Zentraler Mechatronik-Workflow-Kern")
def workflow_kern(rohe_auftrags_daten: dict):
    logger = get_run_logger()
    typ = rohe_auftrags_daten.get("auftrags_typ")
    
    # Standard-Struktur zur Übergabe an den Logging-Task initialisieren
    protokoll_daten = {"teilennummer": rohe_auftrags_daten.get("teilennummer", "UNBEKANNT")}
    fehler_nachricht = None
    
    try:
        # Route 1: 3D-Druck Demonstrator
        if typ == "3d_druck":
            validierter_auftrag = DruckAuftragModel(**rohe_auftrags_daten)
            
            # Module ausführen
            pruef_ergebnis = pruefe_bauteil_und_material(validierter_auftrag)
            gcode_pfad = generiere_gcode_via_orcaslicer(validierter_auftrag, pruef_ergebnis["verfahren"])
            
            # Protokolldaten befüllen
            protokoll_daten.update({
                "teilennummer": validierter_auftrag.teilennummer,
                "verfahren": pruef_ergebnis["verfahren"],
                "nacharbeit": pruef_ergebnis["nacharbeit"],
                "gcode": gcode_pfad
            })
            
        # Route 2: Transferbeispiel Whisper
        elif typ == "audio_transkript":
            validierter_auftrag = AudioAuftragModel(**rohe_auftrags_daten)
            transkript = transkribiere_audio_whisper(validierter_auftrag)
            
            protokoll_daten.update({"verfahren": "Audio-Transkription", "nacharbeit": transkript["text"]})
            
        else:
            raise ValueError(f"Unbekannter Auftragstyp an Systemgrenze abgelehnt: {typ}")
            
    except ValidationError as ve:
        logger.error("-> [VALIDIERUNGSFEHLER] Schnittstellen-Verstoß detektiert!")
        fehler_nachricht = f"Schnittstellenfehler in Feldern: {ve.errors()}"
    except Exception as e:
        logger.error(f"-> [SYSTEMFEHLER] Laufzeitfehler im Modulkern: {str(e)}")
        fehler_nachricht = str(e)
        
    # Die finale Ergebnisprotokollierung wird IMMER ausgeführt (Woche 4 Anforderung)
    schreibe_system_protokoll(typ if typ else "Undefiniert", protokoll_daten, fehler=fehler_nachricht)


# =====================================================================
# PIPELINE-INTEGRATIONSTEST (Musterläufe für Woche 1)
# =====================================================================
if __name__ == "__main__":
    init_datenbank()
    
    # 1. System-Voraussetzungen für den Test simulieren
    with open("bauteil_4711.stl", "w") as f: f.write("solid dummy")
    with open("notiz.wav", "w") as f: f.write("audio dummy")
    if not os.path.exists(ORCA_SLICER_PATH):
        with open(ORCA_SLICER_PATH, "w") as f:
            f.write('#!/bin/bash\necho "OrcaSlicer CLI Mock" \nexit 0')
        subprocess.run(["chmod", "+x", ORCA_SLICER_PATH])

    print("\n--- LAUF 1: Erprobung eines regulären FDM-Druckauftrags ---")
    workflow_kern({
        "auftrags_typ": "3d_druck",
        "teilennummer": "4711",
        "stl_pfad": "bauteil_4711.stl",
        "material_vorgabe": "ASA",
        "toleranz_mm": 0.1
    })

    print("\n--- LAUF 2: Erprobung des Whisper-Transferbeispiels ---")
    workflow_kern({
        "auftrags_typ": "audio_transkript",
        "audio_pfad": "notiz.wav"
    })

    print("\n--- LAUF 3: Erprobung der Absicherung (Falsche Datentypen/Material) ---")
    workflow_kern({
        "auftrags_typ": "3d_druck",
        "teilennummer": "9999",
        "stl_pfad": "bauteil_4711.stl",
        "material_vorgabe": "HOLZ",  # Fehler: Nicht im vordefinierten Pydantic-Literal erlaubt
        "toleranz_mm": 0.1
    })
