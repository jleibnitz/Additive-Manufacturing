2. Der erweiterte Pipeline-Code mit echter OrcaSlicer-Anbindung (`pipeline_slicer.py`)

Wir erweitern das Skript so, dass der Slicer mit einem festen Profil aufgerufen wird, die Ausrichtung vornimmt und den G-Code zur Sichtprüfung ablegt [01.1].

python

```
import os
import subprocess
from pydantic import BaseModel, Field, FilePath
from prefect import task, flow, get_run_logger

# Path zur OrcaSlicer-Ausführungsdatei auf Ihrem Linux-System
# Kann ein AppImage-Pfad oder die installierte Binärdatei sein
ORCA_SLICER_PATH = "./OrcaSlicer.AppImage" 

# =====================================================================
# EXTENSION WOCHE 5: Erweitertes Datenmodell für das Slicen
# =====================================================================

class SlicerAuftragModel(BaseModel):
    """Schnittstelle speziell für den OrcaSlicer-CLI-Task"""
    stl_pfad: FilePath = Field(..., description="Die zu slicende STL-Datei")
    material: str = Field(..., description="Materialvorgabe für das Filament-Profil")
    schichtdicke_mm: float = Field(0.2, description="Gewünschte Schichthöhe")
    support_aktiv: bool = Field(False, description="Stützstruktur generieren")


# =====================================================================
# TASK: Echte OrcaSlicer CLI-Anbindung
# =====================================================================

@task(retries=1, retry_delay_seconds=10)
def fuehre_orcaslicer_aus(auftrag: SlicerAuftragModel, ausgabe_ordner: str = "./druckdateien"):
    logger = get_run_logger()
    logger.info(f"[OrcaSlicer] Starte Slicing-Prozess für: {auftrag.stl_pfad.name}")
    
    # 1. Zielverzeichnis für G-Code erstellen falls nicht vorhanden
    os.makedirs(ausgabe_ordner, exist_ok=True)
    
    # Ausgabepfad definieren (z.B. ./druckdateien/halter_4711.gcode)
    basis_name = os.path.splitext(auftrag.stl_pfad.name)[0]
    gcode_ausgabe_pfad = os.path.join(ausgabe_ordner, f"{basis_name}.gcode")
    
    # 2. CLI-Befehl für den OrcaSlicer zusammenbauen
    # Hinweis: Die genauen Flags hängen von Ihrer OrcaSlicer-Version ab.
    # --slice startet den Slicer headless im CLI-Modus.
    cmd = [
        ORCA_SLICER_PATH,
        "--slice", str(auftrag.stl_pfad),
        "--output", gcode_ausgabe_pfad
    ]
    
    # Optionale Parameter dynamisch hinzufügen (Modul-Flexibilität)
    if auftrag.support_aktiv:
        cmd.append("--enable-support")
        
    # Hier können Sie Ihre vordefinierten Projekt-Profile übergeben
    # cmd.extend(["--load-settings", f"profile_{auftrag.material.lower()}.json"])

    logger.info(f"[OrcaSlicer] Führe Befehl aus: {' '.join(cmd)}")
    
    try:
        # 3. Prozess headless auf Linux ausführen
        # stdout und stderr werden abgefangen, um verständliche Fehlerberichte zu erzeugen
        result = subprocess.run(
            cmd, 
            capture_output=True, 
            text=True, 
            check=True
        )
        
        logger.info("[OrcaSlicer] Slicing erfolgreich abgeschlossen.")
        logger.debug(f"[OrcaSlicer Konsole]: {result.stdout}")
        
        return {
            "status": "Erfolgreich geslict",
            "gcode_pfad": gcode_ausgabe_pfad,
            "slicer_log": result.stdout[:200] + "..." # Gekürzte Konsolenausgabe
        }
        
    except subprocess.CalledProcessError as e:
        logger.error(f"[OrcaSlicer] CLI-Fehler bei der Ausführung!")
        logger.error(f"Slicer-Fehlermeldung: {e.stderr}")
        # Wir werfen den Fehler bewusst weiter, damit Prefect den Task als 'Failed' markiert
        raise RuntimeError(f"OrcaSlicer fehlgeschlagen: {e.stderr}")


# =====================================================================
# CORE WORKFLOW: Einbindung in den Kern
# =====================================================================

@flow(name="OrcaSlicer Pilot Pipeline")
def slicer_pilot_workflow(stl_datei_pfad: str, material: str):
    logger = get_run_logger()
    
    # Konfiguration für das Slicing-Werkzeug vorbereiten
    slicer_daten = SlicerAuftragModel(
        stl_pfad=stl_datei_pfad,
        material=material,
        schichtdicke_mm=0.2,
        support_aktiv=True # Stützen erproben laut Themenentwurf
    )
    
    # Task aufrufen
    slicer_ergebnis = fuehre_orcaslicer_aus(slicer_daten)
    
    logger.info(f"Ergebnis-Pfad zur Sichtprüfung: {slicer_ergebnis['gcode_pfad']}")
    return slicer_ergebnis


# =====================================================================
# SYSTEMTEST (Woche 5 Simulation)
# =====================================================================
if __name__ == "__main__":
    # Dummy-STL erzeugen
    test_stl = "halter_4711.stl"
    with open(test_stl, "w") as f: 
        f.write("solid dummy\nfacet normal 0 0 0\nouter loop\nvertex 0 0 0\nendloop\nendsolid")
        
    # WICHTIG für Ihren Testlauf:
    # Damit dieses Skript durchläuft, muss eine ausführbare OrcaSlicer-Datei 
    # (oder ein valider Dummy wie 'echo "Slicing..." > OrcaSlicer.AppImage') existieren.
    if not os.path.exists(ORCA_SLICER_PATH):
        print(f"Erstelle temporären CLI-Dummy für {ORCA_SLICER_PATH}, da keine echte AppImage gefunden wurde.")
        with open(ORCA_SLICER_PATH, "w") as f:
            f.write('#!/bin/bash\necho "OrcaSlicer CLI Dummy: Slicing abgeschlossen für $3" \necho "Output gespeichert unter $5"')
        subprocess.run(["chmod", "+x", ORCA_SLICER_PATH])

    # Start des Pilotlaufs
    ergebnis = slicer_pilot_workflow(stl_datei_pfad=test_stl, material="ASA")
    print("\nEchter System-Ergebnisbericht für Woche 5:", ergebnis)
```

Wissenschaftlicher Wert für Ihre Auswertung (Woche 7 / 21.–27.12.) [01.2]:

- **Headless-Validierung:** Mit `subprocess.run(..., capture_output=True)` beweisen Sie in Ihrer Studienarbeit, dass die Pipeline vollständig _Headless_ (ohne grafische X11-Oberfläche von Linux) operieren kann [01.1]. Das ist der Kernbeweis für den späteren Einsatz auf dem Headless-Server [01.1].
- **Fehler-Rückkopplung:** Falls der OrcaSlicer eine fehlerhafte oder beschädigte STL-Datei erhält, gibt das CLI einen Exit-Code ungleich 0 zurück. Ihr `subprocess.run(..., check=True)` fängt dies ab, Prefect schreibt das präzise _stderr_ des Slicers in die lokalen Logs [01.1]. Sie erhalten sofort den geforderten „verständlichen Fehlerbericht“ [01.1].
