1. Die Architektur der Protokollierung

Wir nutzen das Standard-Modul `sqlite3`, da es unter Linux ohne zusätzliche Installationen einsatzbereit ist (ideal für den Headless-Server) [01.1]. Jedes Mal, wenn der Prefect-Kern einen Lauf beendet (egal ob erfolgreich oder mit Validierungsfehler), wird ein standardisierter Datensatz geschrieben [01.1].

import os
import csv
import sqlite3
from datetime import datetime
from pydantic import BaseModel, Field
from prefect import task, flow, get_run_logger

DB_PFAD = "pipeline_protokoll.db"
CSV_PFAD = "pipeline_ergebnisse.csv"

# =====================================================================
# WOCHE 4: DATENBANK-INITIALISIERUNG
# =====================================================================
def init_datenbank():
    """Erstellt die lokale SQLite-Tabelle für die Auswertung, falls nicht existent."""
    with sqlite3.connect(DB_PFAD) as conn:
        cursor = conn.cursor()
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS auftrags_protokoll (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp TEXT,
                auftrags_typ TEXT,
                teilennummer TEXT,
                status TEXT,
                verfahren_vorgeschlagen TEXT,
                nacharbeit TEXT,
                fehler_details TEXT
            )
        """)
        conn.commit()


# =====================================================================
# TASK: Lokales Schreiben der Berichte (Zentraler Kern-Bestandteil)
# =====================================================================
@task
def protokolliere_ergebnis_lokal(typ: str, daten: dict, fehler: str = None):
    logger = get_run_logger()
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    
    # Daten für die Zeile aufbereiten
    teilennummer = daten.get("teilennummer", "UNBEKANNT")
    status = "FEHLER" if fehler else "ERFOLGREICH"
    verfahren = daten.get("verfahren", "N/A")
    nacharbeit = daten.get("nacharbeit", "N/A")
    fehler_text = str(fehler) if fehler else ""

    logger.info(f"[Protokoll] Speichere Lauf für Bauteil {teilennummer} ({status})...")

    # 1. In SQLite-Datenbank schreiben
    with sqlite3.connect(DB_PFAD) as conn:
        cursor = conn.cursor()
        cursor.execute("""
            INSERT INTO auftrags_protokoll 
            (timestamp, auftrags_typ, teilennummer, status, verfahren_vorgeschlagen, nacharbeit, fehler_details)
            VALUES (?, ?, ?, ?, ?, ?, ?)
        """, (now, typ, teilennummer, status, verfahren, nacharbeit, fehler_text))
        conn.commit()

    # 2. In CSV-Datei schreiben (Erleichtert Excel/Origin-Import für die Arbeit)
    csv_exists = os.path.exists(CSV_PFAD)
    with open(CSV_PFAD, mode="a", newline="", encoding="utf-8") as csv_file:
        writer = csv.writer(csv_file, delimiter=";")
        if not csv_exists:
            # Header schreiben, falls Datei neu ist
            writer.writerow(["Zeitstempel", "Typ", "Teilenummer", "Status", "Verfahren", "Nacharbeit", "Fehler"])
        writer.writerow([now, typ, teilennummer, status, verfahren, nacharbeit, fehler_text])


# =====================================================================
# SIMULIERTER ARCHITEKTUR-KERN
# =====================================================================
@flow(name="Pipeline mit lokaler Ergebnisdatenbank")
def core_workflow_mit_logging(roher_auftrag: dict):
    logger = get_run_logger()
    typ = roher_auftrag.get("auftrags_typ", "unbekannt")
    
    # Platzhalter für die Validierung aus Woche 3
    fehler_aufgetreten = None
    verarbeitungs_daten = {"teilennummer": roher_auftrag.get("teilennummer", "4711")}
    
    logger.info(f"Verarbeite {verarbeitungs_daten['teilennummer']}...")
    
    # Künstliche Fehler-Simulation für den Testlauf
    if roher_auftrag.get("simuliere_fehler"):
        fehler_aufgetreten = "Schnittstellenfehler: Material 'HOLZ' nicht zulässig."
    else:
        # Erfolgreicher Dummy-Pfad
        verarbeitungs_daten.update({
            "verfahren": "FDM (Vorgeschlagen wegen ASA)",
            "nacharbeit": "M4-Gewinde markiert"
        })

    # Aufruf des Protokoll-Tasks am Ende des Kerns
    protokolliere_ergebnis_lokal(typ, verarbeitungs_daten, fehler=fehler_aufgetreten)
    return {"status": "Protokolliert"}


# =====================================================================
# WERKZEUG-TEST: Automatische Erzeugung der 8 Musteraufträge
# =====================================================================
if __name__ == "__main__":
    init_datenbank()
    
    print("--- Generiere 8 Musterläufe für die Auswertungstabelle ---")
    
    # Wir simulieren 6 erfolgreiche Läufe und 2 Klärfälle/Fehlerläufe
    for i in range(1, 9):
        ist_fehlerlauf = (i in) # Lauf 5 und 8 schlagen fehl
        
        auftrag = {
            "auftrags_typ": "3d_druck",
            "teilennummer": f"Muster_471{i}",
            "simuliere_fehler": ist_fehlerlauf
        }
        
        core_workflow_mit_logging(auftrag)
        
    print("\n--- Auswertung der SQLite-Datenbank für Ihre Studienarbeit ---")
    with sqlite3.connect(DB_PFAD) as conn:
        cursor = conn.cursor()
        cursor.execute("SELECT id, teilennummer, status, verfahren_vorgeschlagen, fehler_details FROM auftrags_protokoll")
        for row in cursor.fetchall():
            print(f"Lauf ID {row[0]} | Bauteil: {row[1]} | Status: {row[2]} | Verfahren: {row[3]} | Fehler: {row[4]}")
            
    print(f"\n[System] Die Daten wurden zusätzlich in '{CSV_PFAD}' für Ihre Grafiken gespeichert.")
